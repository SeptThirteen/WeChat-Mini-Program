flowchart TD
    subgraph TabBar["TabBar 底部导航"]
        INDEX["首页<br/>pages/index/index"]
        PROFILE["个人中心<br/>pages/user/profile"]
    end

    INDEX <-->|TabBar 切换| PROFILE

    subgraph 首页模块
        INDEX -->|点击服务卡片| SVC_DETAIL["服务详情<br/>pages/service/detail"]

    end

    subgraph 个人中心模块
        PROFILE -->|紧急联系人| EMERGENCY["紧急联系人<br/>pages/user/emergency"]

        PROFILE -->|政务代办| GOV["政务代办<br/>pages/user/gov-tasks"]

    end

    end

    LOGIN -->|登录成功| INDEX
